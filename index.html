create table if not exists public.attendance ( id bigint primary key, owner_id uuid not null references auth.users(id) on delete cascade, client_id bigint not null references public.clients(id) on delete cascade, attendance_date date not null, class_time text not null, created_at timestamptz default now() ); alter table public.attendance enable row level security; revoke all on table public.attendance from anon, authenticated; grant select, insert, update, delete on table public.attendance to authenticated; create policy "attendance_select_own" on public.attendance for select to authenticated using ((select auth.uid()) = owner_id); create policy "attendance_insert_own" on public.attendance for insert to authenticated with check ((select auth.uid()) = owner_id); create policy "attendance_update_own" on public.attendance for update to authenticated using ((select auth.uid()) = owner_id) with check ((select auth.uid()) = owner_id); create policy "attendance_delete_own" on public.attendance for delete to authenticated using ((select auth.uid()) = owner_id); create index if not exists attendance_owner_idx on public.attendance(owner_id); create index if not exists attendance_date_class_idx on public.attendance(attendance_date, class_time);

alter table public.clients add column if not exists discount numeric default 0, add column if not exists frozen_until date, add column if not exists linked_client_id bigint references public.clients(id) on delete set null; create table if not exists public.membership_actions ( id bigint primary key, owner_id uuid not null references auth.users(id) on delete cascade, client_id bigint not null references public.clients(id) on delete cascade, action_type text not null, detail text, action_date date not null default current_date, created_at timestamptz default now() ); create table if not exists public.collections ( id bigint primary key, owner_id uuid not null references auth.users(id) on delete cascade, client_id bigint not null references public.clients(id) on delete cascade, status text default 'pendiente', last_contact_date date, promise_date date, notes text, created_at timestamptz default now(), unique(owner_id, client_id) ); create table if not exists public.expenses ( id bigint primary key, owner_id uuid not null references auth.users(id) on delete cascade, category text not null, amount numeric not null, method text, expense_date date not null, notes text, created_at timestamptz default now() ); alter table public.membership_actions enable row level security; alter table public.collections enable row level security; alter table public.expenses enable row level security; revoke all on table public.membership_actions from anon, authenticated; revoke all on table public.collections from anon, authenticated; revoke all on table public.expenses from anon, authenticated; grant select, insert, update, delete on table public.membership_actions to authenticated; grant select, insert, update, delete on table public.collections to authenticated; grant select, insert, update, delete on table public.expenses to authenticated; create policy "membership_actions_own" on public.membership_actions for all to authenticated using ((select auth.uid()) = owner_id) with check ((select auth.uid()) = owner_id); create policy "collections_own" on public.collections for all to authenticated using ((select auth.uid()) = owner_id) with check ((select auth.uid()) = owner_id); create policy "expenses_own" on public.expenses for all to authenticated using ((select auth.uid()) = owner_id) with check ((select auth.uid()) = owner_id); create index if not exists membership_actions_client_idx on public.membership_actions(client_id); create index if not exists collections_client_idx on public.collections(client_id); create index if not exists expenses_date_idx on public.expenses(expense_date);

alter table public.clients add column if not exists photo_path text; insert into storage.buckets (id, name, public) values ('client-photos', 'client-photos', false) on conflict (id) do update set public = false; drop policy if exists "client_photos_select_own" on storage.objects; drop policy if exists "client_photos_insert_own" on storage.objects; drop policy if exists "client_photos_update_own" on storage.objects; drop policy if exists "client_photos_delete_own" on storage.objects; create policy "client_photos_select_own" on storage.objects for select to authenticated using ( bucket_id = 'client-photos' and (storage.foldername(name))[1] = (select auth.uid())::text ); create policy "client_photos_insert_own" on storage.objects for insert to authenticated with check ( bucket_id = 'client-photos' and (storage.foldername(name))[1] = (select auth.uid())::text ); create policy "client_photos_update_own" on storage.objects for update to authenticated using ( bucket_id = 'client-photos' and (storage.foldername(name))[1] = (select auth.uid())::text ) with check ( bucket_id = 'client-photos' and (storage.foldername(name))[1] = (select auth.uid())::text ); create policy "client_photos_delete_own" on storage.objects for delete to authenticated using ( bucket_id = 'client-photos' and (storage.foldername(name))[1] = (select auth.uid())::text );
